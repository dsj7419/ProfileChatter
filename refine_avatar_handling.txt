  8: class TextProcessor {
  9:   /**
 10:    * Calculate approximate text width (reliable enough for node environment)
 11:    * @param {string} text - Text to measure
 12:    * @returns {number} - Estimated pixel width
 13:    */
 14:   measureTextWidth(text) {
 15:     // Simple heuristic based on average character width
 16:     // Different characters have different widths, so this is an approximation
 17:     // For a more precise measurement, we would need a canvas context
 18:     const avgCharWidth = config.layout.FONT_SIZE_PX * 0.6; // Rough approximation for proportional fonts
 19:     return text.length * avgCharWidth;
 20:   }
 21:   
 22:   /**
 23:    * Calculate available width for chat bubbles, accounting for avatars if enabled
 24:    * @returns {number} - Maximum available width for bubbles in pixels
 25:    */
 26:   getAvailableBubbleWidth() {
 27:     let maxWidth = config.layout.MAX_BUBBLE_W_PX;
 28:     
 29:     // If avatars are enabled, reduce available space by avatar width + offset
 30:     if (config.avatars && config.avatars.enabled) {
 31:       maxWidth -= (config.avatars.sizePx + (config.avatars.xOffsetPx * 2));
 32:     }
 33:     
 34:     return maxWidth;
 35:   }
 36:   
 37:   /**
 38:    * Wrap text to fit within chat bubble width constraints, accounting for avatars
 39:    * @param {string} text - Raw text to wrap
 40:    * @param {number} [maxWidth] - Optional maximum width override
 41:    * @returns {Object} - Wrapped lines and calculated dimensions
 42:    */
 43:   wrapText(text, maxWidth) {
 44:     // Handle multi-line input (e.g., from \n in the text)
 45:     const paragraphs = text.split('\n');
 46:     const allLines = [];
 47:     
 48:     // Get adjusted maximum width - use provided maxWidth if available,
 49:     // otherwise calculate based on avatar configuration
 50:     const maxBubbleWidth = maxWidth !== undefined ? maxWidth : this.getAvailableBubbleWidth();
 51:     
 52:     // Process each paragraph 
 53:     for (const paragraph of paragraphs) {
 54:       const words = paragraph.split(/\s+/);
 55:       let currentLine = '';
 56:       
 57:       // Wrap words to fit maximum width
 58:       for (const word of words) {
 59:         // Skip empty words
 60:         if (!word) continue;
 61:         
 62:         const testLine = currentLine ? `${currentLine} ${word}` : word;
 63:         const testWidth = this.measureTextWidth(testLine);
 64:         
 65:         if (testWidth <= maxBubbleWidth - 2 * config.layout.BUBBLE_PAD_X_PX) {
 66:           currentLine = testLine;
 67:         } else {
 68:           if (currentLine) allLines.push(currentLine);
 69:           currentLine = word;
 70:         }
 71:       }
 72:       
 73:       // Add final line from paragraph
 74:       if (currentLine) {
 75:         allLines.push(currentLine);
 76:       }
 77:     }
 78:     
 79:     // Ensure we have at least one line
 80:     if (allLines.length === 0) {
 81:       allLines.push('');
 82:     }
 83:     
 84:     // Calculate maximum width needed
 85:     const maxLineWidth = Math.max(
 86:       ...allLines.map(line => this.measureTextWidth(line)),
 87:       config.layout.MIN_BUBBLE_W_PX - 2 * config.layout.BUBBLE_PAD_X_PX // Ensure minimum width
 88:     );
 89:     
 90:     // Calculate bubble dimensions (respecting the adjusted maximum width)
 91:     const width = Math.min(
 92:       maxLineWidth + 2 * config.layout.BUBBLE_PAD_X_PX,
 93:       maxBubbleWidth
 94:     );
 95:     
 96:     const height = Math.max(
 97:       allLines.length * config.layout.LINE_HEIGHT_PX + 2 * config.layout.BUBBLE_PAD_Y_PX,
 98:       2 * config.layout.BUBBLE_PAD_Y_PX + config.layout.LINE_HEIGHT_PX // Ensure minimum height
 99:     );
100:     
101:     return {
102:       lines: allLines,
103:       width,
104:       height,
105:       lineCount: allLines.length
106:     };
107:   }
108:   
109:   /**
110:    * Escape special XML characters for SVG text
111:    * @param {string} text - Raw text
112:    * @returns {string} - XML-safe text
113:    */
114:   escapeXML(text) {
115:     return text
116:       .replace(/&/g, '&amp;')
117:       .replace(/</g, '&lt;')
118:       .replace(/>/g, '&gt;')
119:       .replace(/"/g, '&quot;')
120:       .replace(/'/g, '&apos;');
121:   }
122: }

 15: function deepMerge(target, source) {
 16:   for (const key in source) {
 17:     if (source[key] && typeof source[key] === 'object' && !Array.isArray(source[key])) {
 18:       if (!target[key]) Object.assign(target, { [key]: {} });
 19:       deepMerge(target[key], source[key]);
 20:     } else {
 21:       Object.assign(target, { [key]: source[key] });
 22:     }
 23:   }
 24:   return target;
 25: }
